export function FieldTable({ fields }) {
  return (
    <div className="mt-4 overflow-x-auto rounded-xl border border-gray-200">
      <table className="w-full border-collapse text-left text-sm">
        <thead>
          <tr className="border-b border-gray-200 bg-gray-50">
            <th className="px-4 py-3 font-semibold">字段</th>
            <th className="px-4 py-3 font-semibold">类型</th>
            <th className="px-4 py-3 font-semibold">必填</th>
            <th className="px-4 py-3 font-semibold">说明</th>
            <th className="px-4 py-3 font-semibold">示例</th>
          </tr>
        </thead>
        <tbody>
          {fields.map((f) => (
            <tr key={f.name} className="border-b border-gray-100 last:border-0">
              <td className="px-4 py-3 font-mono text-xs">{f.name}</td>
              <td className="px-4 py-3">{f.type}</td>
              <td className="px-4 py-3">{f.required ? '是' : '否'}</td>
              <td className="px-4 py-3 text-gray-600">{f.desc}</td>
              <td className="px-4 py-3 font-mono text-xs text-gray-700">{f.example}</td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}
